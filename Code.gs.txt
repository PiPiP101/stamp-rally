/* ================================================================ ENTRY */

function doGet(e) {
  const p = (e && e.parameter) || {};
  if (p.action) {
    if (p.action !== 'ping' && p.action !== 'config') return json_(fail_('METHOD_NOT_ALLOWED', 'คำสั่งนี้ต้องส่งแบบ POST'));
    return json_(route_({ action: p.action }));
  }
  let title = 'Axtra Club Stamp Rally';
  try { title = (getSettings_().EVENT_NAME || 'Axtra Club') + ' · Stamp Rally'; } catch (err) { /* ยังไม่ได้ setup */ }
  // ส่ง URL ของ Web app ให้หน้าเว็บไว้ใช้สำรอง (บางแอปเปิดลิงก์แล้ว google.script.run ใช้ไม่ได้)
  const boot = '<script>window.__EXEC_URL=' + JSON.stringify(serviceUrl_()).replace(/</g, '\\u003c') + ';</script>';
  return HtmlService.createHtmlOutputFromFile('index')
    .append(boot)
    .setTitle(title)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, viewport-fit=cover')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function doPost(e) {
  let req;
  try { req = JSON.parse((e && e.postData && e.postData.contents) || '{}'); }
  catch (err) { return json_(fail_('BAD_JSON', 'รูปแบบข้อมูลไม่ถูกต้อง')); }
  return json_(route_(req));
}

function api(req) { return route_(req); }

function route_(req) {
  req = req || {};
  const action = String(req.action || '');
  try {
    switch (action) {
      case 'ping': return ok_({ time: Date.now() });
      case 'config': return ok_(publicConfig_(getSettings_()));
      case 'scan': return scan_(req);
      case 'state': return state_(req);
      case 'login': return login_(req);
      case 'session': return session_(req);
      case 'dashboard': return dashboard_(req);
      case 'export': return export_(req);
      case 'points': return points_(req);
      case 'simulate': return simulate_(req);
      default: return fail_('UNKNOWN_ACTION', 'ไม่รู้จักคำสั่ง: ' + action);
    }
  } catch (err) {
    console.error(action, err && err.stack ? err.stack : err);
    return fail_('SERVER_ERROR', 'ระบบขัดข้องชั่วคราว กรุณาสแกนอีกครั้งค่ะ', { detail: String(err && err.message ? err.message : err) });
  }
}

/** สแกน QR บูธ (B1–B3) หรือจุดรับของ (EX)
 *  ข้อมูลบัตรอ่าน/เขียนที่ Script Properties ภายใน Lock (เร็ว ~0.2 วินาที) แล้วค่อยเขียนแถวในชีต + บันทึกการสแกนนอก Lock
 *  ทำให้รับคนสแกนพร้อมกันได้มากขึ้นหลายเท่า */
function scan_(req) {
  const pt = parsePoint_(req.s);
  if (!pt) return fail_('BAD_QR', 'QR นี้ใช้ไม่ได้ค่ะ กรุณาสแกนจากป้ายในงาน');
  const ch = req.ch === 'LINE' ? 'LINE' : 'WEB';
  const tel = normTel_(req.tel);
  if (req.tel && !tel) return fail_('BAD_TEL', 'กรุณากรอกเบอร์มือถือ 10 หลัก เช่น 0812345678');
  const code = String(req.code || '').toUpperCase();
  const devOk = !!code && !!req.key && req.key === cardKey_(code);
  // เครื่องไม่มีบัตร และยังไม่กรอกเบอร์ → ขอเบอร์ทันที ไม่ต้องเข้าคิว
  if (!devOk && !tel) return ok_({ result: 'NEED_TEL', point: pt });
  // rid = รหัสของการสแกนครั้งนั้น ถ้าเน็ตหลุดแล้วหน้าเว็บส่งซ้ำ จะได้ผลเดิม (ไม่กลายเป็น "รับไปแล้ว")
  const rid = String(req.rid || '').replace(/[^A-Za-z0-9]/g, '').slice(0, 32);
  const cache = CacheService.getScriptCache(), ridKey = rid.length >= 8 ? 'scan_' + rid : '';
  const replay = function () { const hit = ridKey && cache.get(ridKey); return hit ? JSON.parse(hit) : null; };
  let out = replay();
  if (out) return out;
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(25000)) return fail_('BUSY', 'มีผู้สแกนพร้อมกันจำนวนมาก กรุณาสแกนอีกครั้งค่ะ');
  let result = '', p = null;
  const now = new Date();
  try {
    out = replay();
    if (out) return out;
    p = cardFor_(code, req.key, tel);
    if (!p && !tel) out = ok_({ result: 'NEED_TEL', point: pt });          // บัตรในเครื่องถูกล้างไปแล้ว
    else if (pt === 'EX' && !p) out = ok_({ result: 'NO_CARD', point: pt, tel4: tel.slice(-4) });
    else {
      const pr = props_(), extra = {};
      if (pt === 'EX') {
        if (p.redeemedAt) result = 'ALREADY';
        else if (countStamps_(p) < 3) { p.exitFails = (Number(p.exitFails) || 0) + 1; result = 'INCOMPLETE'; }
        else { p.redeemedAt = now; result = 'REDEEMED'; }
      } else {
        const k = 'b' + pt.charAt(1);
        if (!p) {
          p = blankPlayer_(uniqueCode_(pr), now, ch);
          p.row = Number(pr.getProperty(KEY_NEXT_ROW)) || (sheet_(SHEETS.PLAYERS).getLastRow() + 1);
          extra[KEY_NEXT_ROW] = String(p.row + 1);
        }
        if (p.redeemedAt) result = 'ALREADY';
        else if (p[k]) result = 'DUP';
        else { p[k] = now; if (!p.firstBooth) p.firstBooth = pt; result = 'NEW_STAMP'; }
      }
      if (tel && !p.phoneKey) { p.phoneKey = phoneKey_(tel); p.phone4 = tel.slice(-4); extra[PFX_TEL + p.phoneKey] = p.code; }
      p.lastScanAt = now;
      recompute_(p);
      saveP_(pr, p, extra);
      out = ok_({ result: result, point: pt, card: cardOut_(p) });
      if (ridKey) cache.put(ridKey, JSON.stringify(out), SCAN_REPLAY_SEC);
    }
  } finally { lock.releaseLock(); }
  if (result) writeSheetRow_(p);
  logScan_(now, p ? p.code : '', pt, result || out.data.result, ch);
  return out;
}

/** เปิดบัตรของตัวเอง (ไม่ได้สแกน) */
function state_(req) {
  const tel = normTel_(req.tel);
  if (req.tel && !tel) return fail_('BAD_TEL', 'กรุณากรอกเบอร์มือถือ 10 หลัก เช่น 0812345678');
  const p = cardFor_(req.code, req.key, tel);
  return p ? ok_({ card: cardOut_(p) }) : fail_('NOT_FOUND', tel ? 'ยังไม่มีบัตรของเบอร์นี้ สแกน QR ที่บูธเพื่อเริ่มสะสมได้เลยค่ะ' : 'ไม่พบบัตรในเครื่องนี้');
}

/* ================================================================ DASHBOARD */

function login_(req) {
  const s = getSettings_(), pin = String(req.pin || '').trim();
  if (!s.PIN || pin !== String(s.PIN)) { Utilities.sleep(1200); return fail_('BAD_PIN', 'PIN ไม่ถูกต้อง'); }
  const exp = Date.now() + TOKEN_HOURS * 3600 * 1000;
  return ok_({ token: 'staff.' + exp + '.' + signToken_(exp, s), role: 'staff', expiresAt: exp });
}

function session_(req) { return auth_(req.token) ? ok_({ role: 'staff' }) : fail_('AUTH', 'กรุณาใส่ PIN อีกครั้ง'); }

function dashboard_(req) {
  if (!auth_(req.token)) return fail_('AUTH', 'กรุณาใส่ PIN อีกครั้ง');
  const cache = CacheService.getScriptCache(), hit = cache.get(CACHE_DASH);
  if (hit) return ok_(JSON.parse(hit));
  const s = getSettings_();
  const lite = allPlayers_().filter(function (p) { return !p.mergedInto; }).map(function (p) {
    return { code: p.code, ch: p.channel, b: [ms_(p.b1), ms_(p.b2), ms_(p.b3)], red: ms_(p.redeemedAt), fails: Number(p.exitFails) || 0 };
  });
  const out = {
    summary: buildStampSummary(lite, { boothNames: boothNames_(s), nowMs: Date.now(), startMin: hmToMin_(s.EVENT_START), step: 15 }),
    generatedAt: Date.now(), eventName: s.EVENT_NAME, prototype: isTrue_(s.PROTOTYPE_MODE)
  };
  const txt = JSON.stringify(out);
  if (txt.length < 95000) cache.put(CACHE_DASH, txt, 8);
  return ok_(out);
}

function export_(req) {
  if (!auth_(req.token)) return fail_('AUTH', 'กรุณาใส่ PIN อีกครั้ง');
  const s = getSettings_();
  const rows = allPlayers_().filter(function (p) { return !p.mergedInto; }).map(function (p) {
    const o = {};
    COLS.forEach(function (c) {
      if (c[1] === 'phoneKey') return;   // ไม่ส่งค่าเบอร์เข้ารหัสออกไป
      const v = p[c[1]];
      o[c[1]] = DATE_KEYS.indexOf(c[1]) > -1 ? (toDate_(v) ? Utilities.formatDate(toDate_(v), TZ, 'yyyy-MM-dd HH:mm:ss') : '') : v;
    });
    return o;
  });
  return ok_({ players: rows, boothNames: boothNames_(s), generatedAt: Date.now(), eventName: s.EVENT_NAME });
}

function points_(req) {
  if (!auth_(req.token)) return fail_('AUTH', 'กรุณาใส่ PIN อีกครั้ง');
  const s = getSettings_(), names = boothNames_(s).concat([s.EXIT_NAME || 'จุดรับแลกของที่ระลึก']);
  return ok_({ publicUrl: serviceUrl_(), appUrl: appUrl_(), points: POINTS.map(function (id, i) { return { id: id, name: names[i], s: id + '-' + pointSig_(id) }; }) });
}

/** โหมด Prototype: เพิ่มผู้เล่นจำลองและเดินเกมต่อให้คนที่ยังเล่นไม่ครบ */
function simulate_(req) {
  if (!auth_(req.token)) return fail_('AUTH', 'กรุณาใส่ PIN อีกครั้ง');
  if (!isTrue_(getSettings_().PROTOTYPE_MODE)) return fail_('OFF', 'ปิดโหมด Prototype แล้ว');
  const n = Math.max(0, Math.min(20, Number(req.n) || 0));
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) return fail_('BUSY', 'ระบบกำลังทำงาน');
  const touched = [];
  try {
    const pr = props_(), now = new Date(), map = {};
    const list = allPlayers_().filter(function (p) { return !p.mergedInto && !p.redeemedAt; });
    for (let k = 0; k < Math.ceil(n * 1.3) && list.length; k++) {
      const p = list.splice(Math.floor(Math.random() * list.length), 1)[0];
      const missing = ['b1', 'b2', 'b3'].filter(function (b) { return !p[b]; });
      if (missing.length) p[missing[Math.floor(Math.random() * missing.length)]] = now;
      else if (Math.random() < 0.9) p.redeemedAt = now;
      p.lastScanAt = now; recompute_(p); map[PFX_CARD + p.code] = packP_(p); touched.push(p);
    }
    let next = Number(pr.getProperty(KEY_NEXT_ROW)) || (sheet_(SHEETS.PLAYERS).getLastRow() + 1);
    for (let i = 0; i < n; i++) {
      const p = blankPlayer_(uniqueCode_(pr), now, Math.random() < 0.85 ? 'LINE' : 'WEB');
      const b = 1 + Math.floor(Math.random() * 3);
      p['b' + b] = now; p.firstBooth = 'B' + b; p.lastScanAt = now; p.row = next++; recompute_(p);
      map[PFX_CARD + p.code] = packP_(p); touched.push(p);
    }
    map[KEY_NEXT_ROW] = String(next);
    pr.setProperties(map);
  } finally { lock.releaseLock(); }
  touched.forEach(writeSheetRow_);
  CacheService.getScriptCache().remove(CACHE_DASH);
  return ok_({ added: n });
}

/* ================================================================ HELPERS */

/* ---------------- ที่เก็บข้อมูลบัตร (Script Properties) ----------------
 * p.<รหัส> = ข้อมูลบัตร (JSON)   t.<เบอร์เข้ารหัส> = รหัสบัตร   n = แถวถัดไปในชีต Players */
const PFX_CARD = 'p.', PFX_TEL = 't.', KEY_NEXT_ROW = 'n';
function props_() { return PropertiesService.getScriptProperties(); }

// เก็บแบบย่อ (~100 ไบต์/บัตร) เวลาเป็นวินาทีนับจาก 1 ม.ค. 2026 ฐาน 36  ที่เก็บรับได้ 500 KB ≈ 3,000+ บัตร
const T0 = 1767225600000;
function t36_(v) { const m = ms_(v); return m ? Math.round((m - T0) / 1000).toString(36) : 0; }
function d36_(v) { return v ? new Date(T0 + parseInt(v, 36) * 1000) : ''; }

function packP_(p) {
  return JSON.stringify([p.row || 0, t36_(p.createdAt), p.firstBooth || '', t36_(p.b1), t36_(p.b2), t36_(p.b3), t36_(p.completedAt),
    t36_(p.redeemedAt), p.channel === 'LINE' ? 'L' : 'W', Number(p.exitFails) || 0, p.mergedInto || '', t36_(p.lastScanAt), p.phoneKey || '', p.phone4 || '']);
}

function unpackP_(code, txt) {
  const a = JSON.parse(txt);
  const p = { code: code, row: a[0] || 0, createdAt: d36_(a[1]), firstBooth: a[2] || '', b1: d36_(a[3]), b2: d36_(a[4]), b3: d36_(a[5]),
    stamps: 0, completedAt: d36_(a[6]), redeemedAt: d36_(a[7]), status: '', channel: a[8] === 'L' ? 'LINE' : 'WEB', exitFails: a[9] || 0,
    mergedInto: a[10] || '', lastScanAt: d36_(a[11]), phoneKey: a[12] || '', phone4: a[13] || '' };
  recompute_(p);
  return p;
}

function getP_(pr, code) { if (!code) return null; const t = pr.getProperty(PFX_CARD + code); return t ? unpackP_(code, t) : null; }
function follow_(pr, p) { if (p && p.mergedInto) { const t = getP_(pr, p.mergedInto); if (t) return t; } return p; }

function saveP_(pr, p, extra) {
  const m = extra || {};
  m[PFX_CARD + p.code] = packP_(p);
  pr.setProperties(m);
}

function allPlayers_() {
  const all = props_().getProperties(), out = [];
  Object.keys(all).forEach(function (k) { if (k.indexOf(PFX_CARD) === 0) out.push(unpackP_(k.slice(PFX_CARD.length), all[k])); });
  return out.sort(function (a, b) { return (a.row || 0) - (b.row || 0); });
}

/** บัตรที่เครื่องจำไว้ (รหัส + กุญแจที่ระบบเซ็นให้) */
function findCard_(code, key) {
  code = String(code || '').toUpperCase();
  if (!code || !key || key !== cardKey_(code)) return null;
  const pr = props_();
  return follow_(pr, getP_(pr, code));
}

/** เบอร์มือถือไทย 10 หลัก (ตัดขีด/เว้นวรรค, +66 → 0) */
function normTel_(v) {
  let d = String(v || '').replace(/\D/g, '');
  if (/^66\d{9}$/.test(d)) d = '0' + d.slice(2);
  return /^0\d{9}$/.test(d) ? d : '';
}

/** เก็บเบอร์แบบเข้ารหัสทางเดียว (HMAC) ไม่เก็บเบอร์เต็ม */
function phoneKey_(tel) {
  return Utilities.computeHmacSha256Signature('tel|' + tel, secret_()).slice(0, 12)
    .map(function (b) { return ('0' + (b & 0xff).toString(16)).slice(-2); }).join('');
}

function findByPhone_(tel) {
  const pr = props_(), code = pr.getProperty(PFX_TEL + phoneKey_(tel));
  return code ? follow_(pr, getP_(pr, code)) : null;
}

/** หาบัตร: ถ้ากรอกเบอร์ เบอร์คือตัวจริง (บัตรของเบอร์นั้น หรือบัตรในเครื่องที่ยังไม่ผูกเบอร์) ไม่งั้นใช้บัตรที่เครื่องจำไว้ */
function cardFor_(code, key, tel) {
  const dev = findCard_(code, key);
  if (!tel) return dev;
  const byTel = findByPhone_(tel);
  if (byTel) return byTel;
  return dev && !dev.phoneKey ? dev : null;
}

function uniqueCode_(pr, used) {
  let code;
  do {
    code = CODE_FIRST.charAt(Math.floor(Math.random() * CODE_FIRST.length));
    for (let i = 0; i < 4; i++) code += CODE_REST.charAt(Math.floor(Math.random() * CODE_REST.length));
  } while ((used && used[code]) || (pr && pr.getProperty(PFX_CARD + code)));
  if (used) used[code] = 1;
  return code;
}

/** เขียนแถวของบัตรลงชีต Players (สำเนาของข้อมูลจริง) — ทำนอก Lock */
function writeSheetRow_(p) {
  if (!p || !p.row) return;
  try {
    p = getP_(props_(), p.code) || p;
    const sh = sheet_(SHEETS.PLAYERS), max = sh.getMaxRows();
    if (p.row > max) { const add = Math.max(p.row - max, 500); sh.insertRowsAfter(max, add); formatPlayerColumns_(sh, max + 1, add); }
    sh.getRange(p.row, 1, 1, COLS.length).setValues([toRow_(p)]);
  } catch (err) { console.warn('sheet row write failed', p.code, err); }
}

function blankPlayer_(code, now, ch) {
  return { code: code, createdAt: now, firstBooth: '', b1: '', b2: '', b3: '', stamps: 0, completedAt: '', redeemedAt: '', status: '', channel: ch, exitFails: 0, mergedInto: '', lastScanAt: now, phoneKey: '', phone4: '' };
}


function countStamps_(p) { return ['b1', 'b2', 'b3'].filter(function (k) { return !!toDate_(p[k]); }).length; }

function recompute_(p) {
  const n = countStamps_(p);
  p.stamps = n;

}

/* ---- SHARED: สรุปผล Stamp Rally (ใช้ทั้งฝั่ง Apps Script และหน้าเว็บโหมดตัวอย่าง) ----
 * players: [{ code, ch: 'LINE'|'WEB', b: [ms|null, ms|null, ms|null], red: ms|null, fails: number }]
 * cfg:     { boothNames: [3 ชื่อ], nowMs, startMin, step }
 * เวลาใช้เขตเวลาไทย (UTC+7 ไม่มี Daylight saving)
 */
function buildStampSummary(players, cfg) {
  var step = cfg.step || 15;
  var names = cfg.boothNames || ['บูธที่ 1', 'บูธที่ 2', 'บูธที่ 3'];
  var tod = function (ms) { return Math.floor(((ms / 60000) + 420) % 1440); };
  var hhmm = function (m) { return ('0' + Math.floor(m / 60)).slice(-2) + ':' + ('0' + (m % 60)).slice(-2); };
  var groups = [0, 0, 0, 0], booths = [0, 0, 0], line = 0, fails = 0, durs = [], events = [], firsts = [], reds = [];

  players.forEach(function (p) {
    var ts = (p.b || []).filter(function (t) { return !!t; });
    if (!ts.length) return;
    var cnt = Math.min(ts.length, 3);
    if (p.red) groups[3]++; else if (cnt >= 3) groups[2]++; else groups[cnt - 1]++;
    (p.b || []).forEach(function (t, i) { if (t) { booths[i]++; events.push({ ms: t, type: 'stamp', code: p.code, booth: i }); } });
    if (p.red) { reds.push(p.red); events.push({ ms: p.red, type: 'redeem', code: p.code }); }
    if (p.ch === 'LINE') line++;
    fails += Number(p.fails) || 0;
    var first = Math.min.apply(null, ts);
    firsts.push(first);
    if (cnt >= 3) durs.push((Math.max.apply(null, ts) - first) / 60000);
  });

  var n = firsts.length;
  durs.sort(function (a, b) { return a - b; });
  var median = durs.length ? Math.round(durs.length % 2 ? durs[(durs.length - 1) / 2] : (durs[durs.length / 2 - 1] + durs[durs.length / 2]) / 2) : null;
  var pct = function (a) { return n ? Math.round((a * 1000) / n) / 10 : 0; };

  // timeline ทุก 15 นาที
  var nowMin = cfg.nowMs ? tod(cfg.nowMs) : null;
  var all = firsts.concat(reds).map(tod);
  var lo = Math.min.apply(null, [cfg.startMin != null ? cfg.startMin : 1050].concat(all));
  var live = nowMin != null && nowMin >= lo;
  var hi = Math.max.apply(null, [lo + step].concat(all).concat(live ? [nowMin] : [lo + step * 6]));
  lo = Math.floor(lo / step) * step;
  hi = Math.min(Math.floor(hi / step) * step, 1440 - step);
  var labels = [], newB = [], redB = [];
  for (var m = lo; m <= hi; m += step) { labels.push(hhmm(m)); newB.push(0); redB.push(0); }
  var idx = function (ms) { return Math.max(0, Math.min(labels.length - 1, Math.floor((tod(ms) - lo) / step))); };
  firsts.forEach(function (t) { newB[idx(t)]++; });
  reds.forEach(function (t) { redB[idx(t)]++; });
  var cumP = [], cumR = [], a = 0, z = 0;
  newB.forEach(function (v, i) { a += v; z += redB[i]; cumP.push(a); cumR.push(z); });

  var recentFrom = cfg.nowMs ? cfg.nowMs - 15 * 60000 : Infinity;
  events.sort(function (x, y) { return y.ms - x.ms; });

  return {
    totals: {
      players: n,
      redeemed: groups[3],
      complete: groups[2] + groups[3],
      waiting: groups[2],
      completionPct: pct(groups[2] + groups[3]),
      redeemPct: pct(groups[3]),
      medianMinutes: median,
      exitFails: fails,
      line: line,
      linePct: pct(line),
      newPlayers15: firsts.filter(function (t) { return t > recentFrom; }).length,
      redeemed15: reds.filter(function (t) { return t > recentFrom; }).length
    },
    groups: [
      { key: 'g1', label: 'สแกน 1 บูธ', note: 'แล้วไม่เล่นต่อ', count: groups[0], pct: pct(groups[0]) },
      { key: 'g2', label: 'สแกน 2 บูธ', note: 'แล้วไม่เล่นต่อ', count: groups[1], pct: pct(groups[1]) },
      { key: 'g3', label: 'ครบ 3 บูธ', note: 'แต่ยังไม่แลกของ', count: groups[2], pct: pct(groups[2]) },
      { key: 'g4', label: 'ครบ 3 บูธ', note: 'และแลกของแล้ว', count: groups[3], pct: pct(groups[3]) }
    ],
    booths: names.map(function (nm, i) { return { label: nm, count: booths[i], pct: pct(booths[i]) }; }),
    timeline: { labels: labels, players: cumP, redeemed: cumR, newPlayers: newB, redeemedPer: redB, step: step },
    recent: events.slice(0, 14).map(function (e) {
      return { ms: e.ms, at: hhmm(tod(e.ms)), type: e.type, code: e.code, text: e.type === 'redeem' ? 'รับของที่ระลึกแล้ว' : 'ได้ Stamp ' + names[e.booth] };
    })
  };
}

/** Prototype: ใส่ผู้เล่นตัวอย่าง ~450 คน ย้อนหลัง 95 นาทีจากตอนนี้ ให้ Dashboard ดูสมจริง */

  } finally { lock.releaseLock(); }
  CacheService.getScriptCache().remove(CACHE_DASH);
  Logger.log('ล้างข้อมูลผู้เล่นแล้ว');
}

/** ย้ายข้อมูลจากชีต Players เข้าที่เก็บ (ครั้งแรกหลังอัปเดตระบบ) */
function importSheetToStore_(sh) {
  const pr = props_(), all = pr.getProperties();
  if (Object.keys(all).some(function (k) { return k.indexOf(PFX_CARD) === 0; })) {
    if (!all[KEY_NEXT_ROW]) pr.setProperty(KEY_NEXT_ROW, String(Math.max(2, sh.getLastRow() + 1)));
    return 0;
  }
  const map = {}, list = readPlayersWithRow_(sh);
  list.forEach(function (x) {
    x.p.row = x.row;
    map[PFX_CARD + x.p.code] = packP_(x.p);
    if (x.p.phoneKey && !x.p.mergedInto) map[PFX_TEL + x.p.phoneKey] = x.p.code;
  });
  map[KEY_NEXT_ROW] = String(Math.max(2, sh.getLastRow() + 1));
  pr.setProperties(map);
  return list.length;
}

/** เขียนชีต Players ใหม่ทั้งหมดจากข้อมูลจริง (ใช้ถ้าตัวเลขในชีตไม่ตรงกับ Dashboard) */
function syncSheetFromStore() {
  guardOwner_();
  const sh = sheet_(SHEETS.PLAYERS), list = allPlayers_();
  if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, COLS.length).clearContent();
  list.forEach(function (p) { if (p.row) sh.getRange(p.row, 1, 1, COLS.length).setValues([toRow_(p)]); });
  Logger.log('เขียนชีต Players ใหม่จากข้อมูลจริงแล้ว ' + list.length + ' บัตร');
}
